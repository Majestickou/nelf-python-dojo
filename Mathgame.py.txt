class mathgametime: #WIP

  def addition(self):
    
    if "easy" in self.chosen_difficulty:
      num1 = randint(1, 50)
      num2 = randint(1, 50)
      num3 = 0
    elif "medium" in self.chosen_difficulty:
      num1 = randint(50, 500)
      num2 = randint(50, 500)
      num3 = 0
    elif "hard" in self.chosen_difficulty:
      num1 = randint(500, 1000)
      num2 = randint(500, 1000)
      num3 = randint(100, 200)
    #counting
    solution = num1 + num2 + num3
    if "easy" in self.chosen_difficulty or "medium" in self.chosen_difficulty:
      question = f"What is the answer to {num1} + {num2}"
    else:
      question = f"What is the answer to {num1} + {num2} + {num3}"
    return num1, num2, num3, solution, question


  def subtraction(self):
    if "easy" in self.chosen_difficulty:
      num1 = randint(25, 50)
      num2 = randint(1, 25)
      num3 = 0
    elif "medium" in self.chosen_difficulty:
      num1 = randint(50, 500)
      num2 = randint(50, 500)
      num3 = 0
    elif "hard" in self.chosen_difficulty:
      num1 = randint(500, 1000)
      num2 = randint(500, 1000)
      num3 = randint(100, 500)
    #counting
    solution = (num1 - num2) - num3
    if "easy" in self.chosen_difficulty or "medium" in self.chosen_difficulty:
      question = f"What is the answer to {num1} - {num2}"
    elif "hard" in self.chosen_difficulty:
      question = f"What is the answer to ({num1} - {num2}) - {num3}"
    return num1, num2, num3, solution, question

    
  def multiplication(self):
    if "easy" in self.chosen_difficulty:
      num1 = randint(2, 20)
      num2 = randint(2, 20)
      num3 = 0
    elif "medium" in self.chosen_difficulty:
      num1 = randint(20, 50)
      num2 = randint(20, 50)
      num3 = 0
    elif "hard" in self.chosen_difficulty:
      num1 = randint(50, 200)
      num2 = randint(50, 200)
      num3 = 0
    #counting
    solution = num1 * num2
    question = f"What is the answer to {num1} * {num2}"
    return num1, num2, num3, solution, question

    
  def division(self):
    if "easy" in self.chosen_difficulty:
      ans = randint(2, 20)
      num2 = randint(2, 20)
      num3 = 0
    elif "medium" in self.chosen_difficulty:
      ans = randint(2, 40)
      num2 = randint(2, 40)
      num3 = 0
    elif "hard" in self.chosen_difficulty:
      ans = randint(2, 100)
      num2 = randint(2, 100)
      num3 = 0
    #counting
    num1 = ans * num2
    solution = ans
    question = f"What is the answer to {num1} / {num2}"
    return num1, num2, num3, solution, question

def again():

      print("Subtraction chosen!")

    elif "multiplication" in chosengame or chosengame == "*":

    elif "division" in chosengame or chosengame == "/":
